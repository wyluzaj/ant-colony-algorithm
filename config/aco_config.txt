# Configuration file for PEA4 Ant Colony Optimization.
# Format: key = value
# Lines starting with # are comments.

# Path to the TSP instance file.
instancePath = data/eil51.tsp

# Path to the file with optimal or best-known values.
# Format in this file: instance_name : optimal_cost
optimalPath = config/optimal.txt

# Base directory for detailed, aggregate and history results.
# Per-run results: results/wyniki/<instance>.csv
# Aggregate results: results/aggregate/aco_aggregate.csv
# History results: results/history/<instance>/<instance>_run<run>_...csv
resultsDirectory = results

# Number of ants.
# ants = 0 means ants = number of cities.
ants = 0

# Number of independent algorithm runs for this one config.
# Use several runs because ACO is randomized.
runs = 10

# Random generator seed.
# seed = 0 means random seed at the beginning of every run.
# seed = 12345 means run 1 uses 12345, run 2 uses 12346, etc.
seed = 0

# Maximum running time in seconds.
maxTimeSeconds = 10

# If true, the algorithm stops when relative error <= targetError.
# Requires that optimalPath contains the value for the selected instance.
stopOnTargetError = true

# Target relative error in percent.
# Example: targetError = 20 means 20%.
targetError = 20

# Importance of pheromone.
alpha = 1.0

# Importance of heuristic information, usually from 2 to 5.
beta = 2.0

# Evaporation rate.
# r = 0 means no evaporation, r = 1 means full evaporation.
r = 0.5

# Initial pheromone value on every edge.
# initialPheromone = 0 means automatic value: ants / Cnn,
# where Cnn is the best nearest-neighbour tour cost.
initialPheromone = 0

# Optional local search improvement after a complete tour is built by an ant.
# useTwoOpt = false means pure ACO.
# useTwoOpt = true means ACO + 2-opt local improvement.
useTwoOpt = false

# Pheromone deposit coefficient Q.
# DAS: delta = Q, deposited after each move.
# QAS: delta = Q / distance(i, j), deposited after each move.
# CAS: delta = Q / tourCost, deposited after a complete tour.
# This is not the initial pheromone value.
depositAmount = 1.0

# Pheromone update mode.
# Deposit timing is fixed by mode and is not configured separately:
# DAS - density ant system, constant pheromone amount Q, deposited after each move.
# QAS - quantity ant system, Q divided by edge length, deposited after each move.
# CAS - cycle ant system, Q divided by complete tour length, deposited after a complete tour.
mode = CAS
